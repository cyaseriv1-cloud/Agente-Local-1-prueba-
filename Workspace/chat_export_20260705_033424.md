% Implementación básica del filtro de Kalman 1D en MATLAB

% Parámetros iniciales
A = 1; % Factor de propagación
H = 1; % Función de medición
Q = 0.1; % Ruido de proceso
R = 0.5; % Ruido de medición
x_hat = 0; % Estado inicial estimado
P = 1; % Covarianza inicial

% Generación de datos simulados
n = 100;
t = 1:n;
u = zeros(1, n);
z = zeros(1, n);

for k = 2:n
    x_true = A * x_true + u(k); % Simulación del estado verdadero
    z(k) = H * x_true + randn; % Medición con ruido
end

% Algoritmo de filtro de Kalman
x_estimates = zeros(1, n);
P_estimates = zeros(1, n);

for k = 2:n
    % Predicción
    x_hat_minus = A * x_hat;
    P_minus = A * P * A' + Q;

    % Actualización
    K = P_minus / (P_minus + R); % Factor de ganancia
    x_hat = x_hat_minus + K * (z(k) - H * x_hat_minus);
    P = (1 - K * H) * P_minus;
    
    % Almacenamiento de resultados
    x_estimates(k) = x_hat;
    P_estimates(k) = P;
end

% Gráfica de resultados
figure;
plot(t, z, 'b.', 'DisplayName', 'Mediciones');
hold on;
plot(t, u, 'g-', 'DisplayName', 'Ruido de proceso');
plot(t, x_true, 'r--', 'DisplayName', 'Estado verdadero');
plot(t, x_estimates, 'm-', 'DisplayName', 'Estimación Kalman');
legend('show');
title('Filtro de Kalman 1D');
xlabel('Tiempo');
ylabel('Valor');
```
